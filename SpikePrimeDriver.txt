from spike import PrimeHub, MotorPair, DistanceSensor
from spike.control import wait_for_seconds

class SpikePrimeDriver:
    def __init__(self, motor_ports=('A', 'B'), distance_port='C', wheel_diameter_cm=5.6):
        self.hub = PrimeHub()
        self.motors = MotorPair(*motor_ports)
        self.distance_sensor = DistanceSensor(distance_port)
        self.wheel_diameter_cm = wheel_diameter_cm
        self.wheel_circumference = 3.14 * wheel_diameter_cm

    def drive_straight(self, distance_cm, speed=50, correction_factor=1.5, stop_distance_cm=10, reverse=False):
        """
        Drives straight using gyro correction and stops if an obstacle is detected.

        Parameters:
        - distance_cm: Distance to travel in centimeters.
        - speed: Motor speed (default is 50).
        - correction_factor: Proportional factor for gyro correction (default is 1.5).
        - stop_distance_cm: Minimum distance to obstacle before stopping (default is 10 cm).
        - reverse: If True, drives backward.
        """
        self.hub.motion_sensor.reset_yaw_angle()
        rotations_needed = distance_cm / self.wheel_circumference

        direction = -1 if reverse else 1
        self.motors.set_default_speed(speed * direction)
        self.motors.pair_motor_positions()
        rotation_count = 0

        while abs(rotation_count) < rotations_needed:
            if not reverse and self.distance_sensor.get_distance_cm() < stop_distance_cm:
                print("Obstacle detected! Stopping.")
                break

            current_heading = self.hub.motion_sensor.get_yaw_angle()
            correction = current_heading * correction_factor * direction
            self.motors.start_tank(speed * direction - correction, speed * direction + correction)
            wait_for_seconds(0.1)
            rotation_count = self.motors.get_position() / 360

        self.motors.stop()

    def turn_by_angle(self, target_angle, turn_speed=30):
        """
        Turns the robot by a specific angle using the gyro sensor.
        Positive = right turn, Negative = left turn.
        """
        self.hub.motion_sensor.reset_yaw_angle()
        current_angle = 0

        if target_angle > 0:
            self.motors.start_tank(turn_speed, -turn_speed)
        else:
            self.motors.start_tank(-turn_speed, turn_speed)

        while abs(current_angle) < abs(target_angle):
            current_angle = self.hub.motion_sensor.get_yaw_angle()
            wait_for_seconds(0.05)

        self.motors.stop()
