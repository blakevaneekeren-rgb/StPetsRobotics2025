wheel_circumference = math.pi * wheel_diameter_cm



def drive_straight(distance_cm, speed, reverse,correction_factor=default_correction_factor, acceleration=200):

    motion_sensor.reset_yaw(0)
    time.sleep(.25)

    rotations_needed = distance_cm / wheel_circumference

    direction = -1 if reverse else 1
    #motors.set_default_speed(speed * direction)
    motor.reset_relative_position(port.C,0)


    rotation_count = 0
    current_speed = speed
    while abs(rotation_count) < rotations_needed:
        if( speed>=500 ):
            if( reverse) :
                if abs(rotation_count) >= abs(.5 * (rotations_needed)) :
                    current_speed = .05 * speed
            else :
                if abs(rotation_count) >= abs(.8 * (rotations_needed)):
                    current_speed = .6 * speed
        else :
            current_speed = speed

        current_heading = motion_sensor.tilt_angles()[0]
        correction = current_heading * correction_factor
        motor_pair.move_tank(motor_pair.PAIR_1, int(current_speed * direction + correction), int(current_speed * direction - correction), acceleration=acceleration)
        time.sleep(.025)
        rotation_count = motor.relative_position(port.C) / 360
        if( log_level >= 7 ) :
            print("current",abs(rotation_count),"out of needed",rotations_needed, "at the speed of", current_speed)
    motor_pair.stop(motor_pair.PAIR_1)
    #if log_level >= 1:
        #print("Reached the destination, ending drive_straight(", distance_cm, ",", speed, ",", reverse, ",", correction_factor, ",", acceleration,")")
    time.sleep(.5)
    log_waypoints()

def turn_by_angle(target_angle, turn_speed):
        motion_sensor.reset_yaw(0)
        time.sleep(.25)
        current_angle = 0
        if target_angle > 0:
            motor_pair.move_tank(motor_pair.PAIR_1, turn_speed, -turn_speed,acceleration=200)
        else:
            motor_pair.move_tank(motor_pair.PAIR_1, -turn_speed, turn_speed,acceleration=200)

        while abs(current_angle) < abs(target_angle):
            current_angle = motion_sensor.tilt_angles()[0] / 10
            #time.sleep(.001)
        if log_level >= 7:
            print("Done Turning to",target_angle,"we are at", -1 * current_angle)
        #if log_level >= 1:
            #print("Reached the destination, ending turn_by_angle(", target_angle,",", turn_speed, ")")
        motor_pair.stop(motor_pair.PAIR_1)
        time.sleep(.5)
        log_waypoints()


def drive_attachment_motor(degrees, speed, move_type):
    motor.reset_relative_position(port.E,0)
    motor.reset_relative_position(port.B,0)

    relative_position = 0

    if move_type == "LEFT":
        motor.run_for_degrees(port.E, degrees, speed)

    elif move_type == "RIGHT":
        motor.run_for_degrees(port.B, degrees, speed)

    elif move_type == "BOTH":
        motor_pair.move_tank(motor_pair.PAIR_2, speed, speed)

        while abs(relative_position) < degrees:
            relative_position = motor.relative_position(port.B)
            if( log_level >= 7 ) :
                print("current",abs(relative_position),"out of needed",degrees, "at the speed of", speed)
        motor_pair.stop(motor_pair.PAIR_2)
    #if log_level >= 1:
        #print("Reached the destination, ending (", distance_cm, ",", speed, ",", reverse, ",", correction_factor, ",", acceleration,")")
    time.sleep(.5)
    log_waypoints()

def drive_forward(distance_cm,speed):
    drive_straight(distance_cm, speed, False, default_correction_factor)
    if log_level >= 1:
        print("Reached the destination, ending drive_forward(", distance_cm, ",", speed, ")")

def drive_forward_until_color(speed, color_needed):

        while color_sensor.color(port.A) != color_needed:
                motor_pair.move_tank(motor_pair.PAIR_1, 200, 200)
                print(color_sensor.color(port.A))
        
        motor_pair.stop(motor_pair.PAIR_1)
        if log_level >= 1:
            print("Reached the destination, ending drive_forward_until_color(", speed, ",", color_needed, ")")
                    

def drive_backwards(distance_cm,speed):
